%=========================================================================
%  BEWERBUNGSDOSSIER (Lebenslauf + Anschreiben + Motivation)
%
%  Gestaltung:  templates/themes/<theme>.tex
%  Inhalt CV:   templates/themes/cv-content.tex
%  Briefe:      templates/cover-letter.tex + motivation.tex (Arbeitskopie)
%
%  Bauen:
%     make application                  aktuelle Bewerbung, Briefe aus templates/
%     make rebuild APP=2026-01-muster-ag archivierte Bewerbung neu bauen
%     make cv                           nur der Lebenslauf
%     make themes                       dasselbe Dossier in allen Themes
%=========================================================================

\documentclass[10pt, a4paper]{article}

%-------------------------------------------------------------------------
%  Gestaltung — hier wechselt man das Theme.
%  Verfügbar unter templates/themes/: kompakt | sidebar | akzent | rueckgrat
%  (\providecommand, damit "make themes" es von aussen setzen kann)
%-------------------------------------------------------------------------

\providecommand{\cvtheme}{kompakt}

%-------------------------------------------------------------------------
%  Zielbewerbung — Ordnername unter applications/.
%  meta.tex liefert \cvjobtitle, \companyname, \positionname, ...
%-------------------------------------------------------------------------

\newcommand{\application}{2026-01-muster-ag}
% Von "make rebuild APP=..." auf der Kommandozeile überschrieben
\ifdefined\appoverride\let\application\appoverride\fi

%-------------------------------------------------------------------------
%  Schalter. Die meta.tex der Bewerbung steuert damit, was erscheint:
%     \cvshow{schluessel}   Baustein dazu, der sonst wegbleibt
%     \cvhide{schluessel}   Baustein weg, der sonst erscheint
%  Die Schlüssel stehen im Kopf von themes/cv-content.tex. Dazu kommen
%     \newcommand{\noletters}{}      Dossier ohne Briefe
%     \newcommand{\nomotivation}{}   ohne Motivationsschreiben
%     \newcommand{\lettersfirst}{}   Briefe vor dem Lebenslauf
%     \cvshow{coverpage}             Deckblatt als erste Seite
%  Beide Makros müssen vor der meta.tex definiert sein.
%-------------------------------------------------------------------------
\providecommand{\cvshow}[1]{\expandafter\def\csname cvshow@#1\endcsname{}}
\providecommand{\cvhide}[1]{\expandafter\def\csname cvhide@#1\endcsname{}}
\input{applications/\application/meta}
\ifdefined\noletters\def\cvonly{}\fi

% Fallback für meta.tex ohne \cvjobtitle
\providecommand{\cvjobtitle}{Entwicklungsingenieur}

% Inhaltsdatei und Sprache — beide kommen aus meta.tex, weil sie zur
% Zielbewerbung gehören und nicht zur Gestaltung. Voreinstellung ist der
% deutsche Inhalt; eine englische Ausschreibung setzt in ihrer meta.tex
%     \newcommand{\cvcontent}{cv-content-en}
%     \newcommand{\cvlanguage}{british}
\providecommand{\cvcontent}{cv-content}

\input{templates/themes/\cvtheme}
\input{templates/themes/\cvcontent}

%-------------------------------------------------------------------------
%  Briefe — normalerweise die Arbeitskopie in templates/. Mit
%  \def\lettersfromapp{} auf der Kommandozeile kommen sie stattdessen aus
%  dem Bewerbungsordner, sodass sich jede archivierte Bewerbung so neu
%  bauen lässt, wie sie abgelegt wurde (siehe "make rebuild").
%-------------------------------------------------------------------------

\ifdefined\lettersfromapp
    \newcommand{\letterdir}{applications/\application}
\else
    \newcommand{\letterdir}{templates}
\fi

% Die Briefe stehen in einer Gruppe, damit Briefschrift und Absatzabstand
% nicht in den Lebenslauf laufen.
% \coveronly und \motivationonly (von "make dossier" gesetzt) bauen nur einen der Briefe.
\newcommand{\dossierletters}{%
    \begingroup
    \ifdefined\motivationonly\else
        \makeletterhead
        \input{\letterdir/cover-letter}
    \fi
    \ifdefined\nomotivation\else\ifdefined\coveronly\else
        \ifdefined\motivationonly\else\newpage\fi
        \makeletterhead
        \input{\letterdir/motivation}
    \fi\fi
    \newpage
    \endgroup
}

% Inhaltsverzeichnis des Deckblatts, in der Reihenfolge des Dossiers
\newcommand{\coverletters}{\LblCoverLetter\ifdefined\nomotivation\else\ \textbullet\ \LblMotivation\fi}
\ifdefined\cvonly
    \renewcommand{\coverpagecontents}{\LblCV}
\else\ifdefined\lettersfirst
    \renewcommand{\coverpagecontents}{\coverletters\ \textbullet\ \LblCV}
\else
    \renewcommand{\coverpagecontents}{\LblCV\ \textbullet\ \coverletters}
\fi\fi

\begin{document}

\ifdefined\letteronly\else\cvopt{coverpage}{\begingroup\makecoverpage\endgroup}\fi

\ifdefined\letteronly
    % nur die Briefe, ohne Lebenslauf (von "make dossier" gesetzt)
    \dossierletters
\else\ifdefined\cvonly
    \makecvbody
\else\ifdefined\lettersfirst
    \dossierletters
    % Die Kopfzeile des Lebenslaufs zählt ab seinem Beginn
    \setcounter{page}{1}
    \makecvbody
\else
    \makecvbody
    \newpage
    \dossierletters
\fi\fi\fi

\end{document}
