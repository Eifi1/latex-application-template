\newcommand{\cvsection}[1]{%
    \par\addvspace{12pt}%
    \Needspace*{5\baselineskip}%
    % Überschrift und Linie sitzen bündig zur Inhaltsspalte (3,0 cm), damit
    % die Zeitachse frei bleibt. Über die volle Breite gesetzt würden beide
    % die Achse kreuzen — das liest sich als Fehler, nicht als Gestaltung.
    \noindent\hspace*{3.0cm}{\mediumfont\fontsize{9.5pt}{9.5pt}\selectfont
        \addfontfeature{LetterSpace=18}\color{ink}\MakeUppercase{#1}}%
    \par\vspace{5pt}%
    \noindent\hspace*{3.0cm}{\color{accentdark}\rule{1.5cm}{1.4pt}}%
    {\color{rule}\rule{\dimexpr\linewidth-4.5cm\relax}{0.5pt}}%
    \par\addvspace{7pt}%
}

% Punkt auf der Schiene, auf Höhe der ersten Titelzeile.
% \leavevmode ist zwingend: ohne es landet die \llap-Box im vertikalen
% Modus, also vor dem \leftskip der Liste — und damit am linken Satzrand
% statt auf der Achse.
\newcommand{\railmark}{%
    \leavevmode
    \llap{\raisebox{0.24em}{%
        \begin{tikzpicture}
            \fill[white] (0,0) circle (4.2pt);
            \fill[accentdark] (0,0) circle (2.6pt);
        \end{tikzpicture}}%
        \hspace{0.45cm}}%
}

% Kontaktblock im Kopfband. Bewusst eine Tabelle und keine Minipages mit
% festen Spaltenanteilen: die Spalten richten sich nach der längsten
% Angabe. Sonst muss bei jeder neuen Adresse, Nummer oder E-Mail neu
% gerechnet werden, und eine längere Adresse bricht nicht um.
\newcommand{\bandcell}[2]{{\color{accent}#1}~~#2}

% Zwei Spalten, drei Zeilen: links die Erreichbarkeit, rechts die Angaben
% zur Person. Drei Spalten nebeneinander passen nicht neben das Portrait —
% die längste Zeile (Adresse) und die E-Mail liegen so untereinander und
% teilen sich dieselbe Spaltenbreite.
\newcommand{\bandcontacts}{%
    {\metafont\color{onband}\renewcommand{\arraystretch}{1.35}%
    \begin{tabular}{@{}l@{\hspace{0.9cm}}l@{}}
        \bandcell{\icoloc}{\CVaddress} &
        \bandcell{\icocal}{\CVbirth} \\
        \bandcell{\icophone}{\CVphone} &
        \bandcell{\icoheart}{\CVcivil} \\
        \bandcell{\icomail}{\href{mailto:\CVmail}{\color{onband}\CVmail}} &
        \bandcell{\CVlinkicon}{\cvlink} \\
    \end{tabular}}%
}

\newcommand{\makecvheader}{%
    % Vollflächiges Band, blutet über die Papierkante hinaus
    \begin{tikzpicture}[remember picture, overlay]
        \fill[band] ($(current page.north west)$)
            rectangle ($(current page.north east)+(0,-6.1cm)$);
    \end{tikzpicture}%
    \vspace*{0.75cm}%
    \noindent
    \begin{minipage}[c]{\dimexpr\linewidth-4.4cm\relax}
        {\namefont\nohyph\color{white}\CVname}\par\vspace{7pt}
        {\rolelabelfont\color{accent}\MakeUppercase{\CVrole}}\par\vspace{13pt}
        {\color{onband!45}\rule{\linewidth}{0.5pt}}\par\vspace{7pt}
        \bandcontacts
    \end{minipage}%
    \hfill
    \begin{minipage}[c]{3.9cm}
        \raggedleft\photobox{3.9cm}{6pt}
    \end{minipage}%
    \par\vspace{0.95cm}%
    % Kurzprofil
    \noindent
    \begin{minipage}[t]{2.2cm}\raggedleft\vtopsafe
        \datefont\color{muted}\strut Kurzprofil\end{minipage}%
    \hspace{0.8cm}%
    \begin{minipage}[t]{\dimexpr\linewidth-3.0cm\relax}%
        \vtopsafe\bodyflow\fontsize{9.4pt}{13.2pt}\selectfont\strut\ContentSummary
    \end{minipage}\par
}

    \fancyhf{}
    \renewcommand{\headrulewidth}{0pt}

        \begin{tikzpicture}[remember picture, overlay]

                \draw[rail, line width=1pt]
                    ($(current page.north west)+(\railx,-\railtop)$) --
                    ($(current page.south west)+(\railx,1.4cm)$);
            \else
                \fill[band] (current page.north west)
                    rectangle ($(current page.north east)+(0,-1.15cm)$);
                \node[anchor=west, inner sep=0]
                    at ($(current page.north west)+(1.6cm,-0.575cm)$)
                    {\mediumfont\fontsize{9pt}{9pt}\selectfont\color{white}\CVname};
                \node[anchor=east, inner sep=0]
                    at ($(current page.north east)+(-1.6cm,-0.575cm)$)
                    {\metafont\color{onband}\LblCV~~\textbullet~~\LblPage\ \thepage};
                \draw[rail, line width=1pt]
                    ($(current page.north west)+(\railx,-1.9cm)$) --
                    ($(current page.south west)+(\railx,1.4cm)$);
            \fi
        \end{tikzpicture}}

\newcommand{\makeletterhead}{%
    \pagestyle{letterpage}\thispagestyle{letterpage}%
    \setlength{\parskip}{7.5pt}%
    \begin{tikzpicture}[remember picture, overlay]
        \fill[band] (current page.north west)
            rectangle ($(current page.north east)+(0,-3.2cm)$);
    \end{tikzpicture}%
    \vspace*{0.35cm}
    \noindent
    \begin{minipage}[b]{\dimexpr\linewidth-5.6cm\relax}
        {\displayfont\fontsize{21pt}{21pt}\selectfont
         \addfontfeature{LetterSpace=-1}\color{white}\CVname}\par\vspace{5pt}
        {\mediumfont\fontsize{8.8pt}{11pt}\selectfont
         \addfontfeature{LetterSpace=16}\color{accent}\MakeUppercase{\CVrole}}
    \end{minipage}%
    \hfill
    \begin{minipage}[b]{5.4cm}
        \raggedleft\metafont\color{onband}\setlength{\parskip}{0pt}
        \CVaddress\par\CVphone\par\href{mailto:\CVmail}{\color{onband}\CVmail}
    \end{minipage}
    \par\vspace{1.55cm}
}
